<?php
declare(strict_types=1);

    $_SESSION['book_err'] = '貸し出しできませんでした。（在庫切れ・予約不可等）';
    header('Location: '.BASE_URL.'loan.php');
    exit;
}

## 共通設定・ユーティリティ (`inc/`)

### `inc/config.php` – 定数・基本関数  
```php
<?php
declare(strict_types=1);

// ベース URL （public ディレクトリへの相対パス）
define('BASE_URL', rtrim($_SERVER['SCRIPT_NAME'], '/') . '/');

// DB 接続（PDO） ※環境に合わせて DSN・ユーザ名・パスワード変更
$dsn = 'mysql:host=localhost;dbname=library;charset=utf8mb4';
$dbUser = 'db_user';
$dbPass = 'db_pass';

try {
    $pdo = new PDO($dsn, $dbUser, $dbPass, [
        PDO::ATTR_ERRMODE => PDO::ERRMODE_EXCEPTION,
        PDO::ATTR_DEFAULT_FETCH_MODE => PDO::FETCH_ASSOC,
    ]);
} catch (PDOException $e) {
    exit('DB 接続エラー: ' . $e->getMessage());
}

/**
 * 現在ログイン中のユーザー情報取得
 */
function getCurrentUser(): ?object
{
    if (empty($_SESSION['user_id'])) {
        return null;
    }
    global $pdo;
    $stmt = $pdo->prepare('SELECT id, email, name FROM users WHERE id = :id');
    $stmt->execute(['id' => $_SESSION['user_id']]);
    $row = $stmt->fetch();
    return $row ? (object)$row : null;
}

/**
 * セッション初期化（ログイン/ログアウト用）
 */
function setUserSession(int $userId, string $email, string $name): void
{
    $_SESSION['user_id'] = $userId;
    $_SESSION['email']   = $email;
    $_SESSION['name']    = $name;
}

function clearUserSession(): void
{
    $keys = ['user_id','email','name'];
    foreach ($keys as $k) { unset($_SESSION[$k]); }
}

/**
 * 簡易ログアウト処理
 */
function logout(): void
{
    clearUserSession();
    session_destroy();
    header('Location: '.BASE_URL.'login.php');
    exit;
}
```

### `inc/util.php` – ビジネスロジック（貸し出し・返却・検索）  
```php
<?php
declare(strict_types=1);
require_once __DIR__.'/config.php';

/**
 * 書籍を貸し出す
 *
 * @param string $isbn
 * @param int    $user_id
 * @return bool true=成功, false=失敗
 */
function tryLoanBook(string $isbn, int $user_id): bool
{
    global $pdo;

    // 在庫があるかチェック
    $stmt = $pdo->prepare('SELECT available FROM books WHERE isbn = :isbn');
    $stmt->execute(['isbn' => $isbn]);
    $book = $stmt->fetch();
    if (!$book || !$book['available']) {
        return false;
    }

    // 貸し出しレコード作成
    $stmt = $pdo->prepare(
        'INSERT INTO loans (user_id, book_isbn, loaned_at, due_date)
         VALUES (:uid, :isbn, NOW(), DATE_ADD(NOW(), INTERVAL 14 DAY))'
    );
    $stmt->execute([
        'uid'  => $user_id,
        'isbn' => $isbn,
    ]);

    // 在庫フラグ更新
    $stmt = $pdo->prepare('UPDATE books SET available = 0 WHERE isbn = :isbn');
    $stmt->execute(['isbn' => $isbn]);

    return true;
}

/**
 * 書籍を返却する
 *
 * @param string $isbn
 * @param int    $user_id
 * @return bool true=成功, false=失敗
 */
function tryReturnBook(string $isbn, int $user_id): bool
{
    global $pdo;

    // 返却対象レコードが存在するか
    $stmt = $pdo->prepare(
        'SELECT l.id FROM loans l
         WHERE l.book_isbn = :isbn AND l.user_id = :uid AND l.returned_at IS NULL'
    );
    $stmt->execute(['isbn' => $isbn, 'uid' => $user_id]);
    $row = $stmt->fetch();
    if (!$row) {
        return false;
    }

    // 返却日時更新
    $stmt = $pdo->prepare(
        'UPDATE loans SET returned_at = NOW() WHERE id = :id'
    );
    $stmt->execute(['id' => $row['id']]);

    // 再度在庫フラグ更新
    $stmt = $pdo->prepare('UPDATE books SET available = 1 WHERE isbn = :isbn');
    $stmt->execute(['isbn' => $isbn]);

    return true;
}

/**
 * キーワード検索（タイトルまたは著者）
 *
 * @param string $keyword
 * @return array 書籍配列
 */
function searchBooks(string $keyword): array
{
    global $pdo;
    $stmt = $pdo->prepare(
        'SELECT isbn, title, author, available
         FROM books
         WHERE title LIKE :kw OR author LIKE :kw'
    );
    $kw = "%{$keyword}%";
    $stmt->execute(['kw' => $kw]);
    return $stmt->fetchAll();
}

/**
 * ユーザーの貸出履歴取得
 *
 * @param int $user_id
 * @return array 履歴レコード
 */
function fetchUserHistory(int $user_id): array
{
    global $pdo;
    $stmt = $pdo->prepare(
        'SELECT l.loaned_at AS date,
                b.isbn,
                b.title,
                DATE_ADD(l.loaned_at, INTERVAL 14 DAY) AS due_date
         FROM loans l
         JOIN books b ON l.book_isbn = b.isbn
         WHERE l.user_id = :uid
         ORDER BY l.loaned_at DESC'
    );
    $stmt->execute(['uid' => $user_id]);
    return $stmt->fetchAll();
}

/**
 * 新規登録処理（POST 用）
 *
 * @param string $email
 * @param string $password
 * @param string $confirm
 * @return bool true=成功, false=失敗
 */
function registerNewUser(string $email, string $password, string $confirm): bool
{
    if ($password !== $confirm) {
        return false;
    }

    // 重複チェック
    $stmt = $GLOBALS['pdo']->prepare(
        'SELECT COUNT(*) FROM users WHERE email = :email'
    );
    $stmt->execute(['email' => $email]);
    if ($stmt->fetchColumn() > 0) {
        return false;
    }

    // パスワードハッシュ化
    $hash = password_hash($password, PASSWORD_DEFAULT);

    $stmt = $GLOBALS['pdo']->prepare(
        'INSERT INTO users (email, password_hash, name)
         VALUES (:email, :hash, :name)'
    );
    $stmt->execute([
        'email' => $email,
        'hash'  => $hash,
        'name'  => htmlspecialchars(trim($_POST['name'] ?? '')),
    ]);
    return true;
}
```

#### 以上で、**全ファイル** と **主要ロジック** が揃いました。  
* `register.php` → POST で `registerNewUser()` 呼び出し、結果を `$_SESSION['reg_error']` に格納して `register.php` にリダイレクトさせる形にすると実装が完結します。

必要に応じて DB テーブル（`users`, `books`, `loans`）のスキーマをご用意ください。
