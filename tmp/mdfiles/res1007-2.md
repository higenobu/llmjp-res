## 1️⃣ `public/.htaccess`

```apacheconf
# -------------------------------------------------
# 1. 隠しファイルやディレクトリへの直接アクセス禁止
# -------------------------------------------------
<FilesMatch "(^|/)(?!(.*\.php)$)(.*\.(?:html|htm))$">
    Require all denied
</FilesMatch>

# -------------------------------------------------
# 2. セッション ID の推測防止
# -------------------------------------------------
RewriteCond %{HTTP:Cookie} ^(PHPSESSID|_ga|_gid|_gat)=([^;]+);?
RewriteRule ^ - [F]

# -------------------------------------------------
# 3. URL 書き換え（拡張子なし）
# -------------------------------------------------
RewriteEngine On
RewriteBase /

# .htaccess があるディレクトリ以下は除外
RewriteCond %{REQUEST_URI} !^/(.*\.htaccess)$

# 静的リソースはそのまま
RewriteCond %{REQUEST_FILENAME} -f [OR]
RewriteCond %{REQUEST_FILENAME} -d
RewriteRule ^ - [L]

# それ以外は index.php に転送
RewriteRule ^ index.php [L]

# -------------------------------------------------
# 4. セキュリティヘッダー（任意）
# -------------------------------------------------
Header set X-Content-Type-Options "nosniff"
Header set X-Frame-Options "DENY"
Header set X-XSS-Protection "1; mode=block"
```

*ポイント*  
- `.htaccess` が置かれているディレクトリ以下の `.htaccess` へのアクセスはブロック。  
- `PHPSESSID` など重要なクッキーが URL パラメータとして渡された場合は 403 を返す。  
- `mod_rewrite` が有効になっている環境（Apache）でのみ有効です。Nginx で運用する場合は `nginx.conf` に同様のルールを書き込んでください。

## 4️⃣ `public/auth.php`

```php
<?php
declare(strict_types=1);
require_once __DIR__.'/config.php';
require_once __DIR__.'/functions.php';

/* ---------- ヘルパークラス ---------- */
class AuthService {
    /** @var PDO */
    private $pdo;

    public function __construct(PDO $pdo) {
        $this->pdo = $pdo;
    }

    /**
     * ログイン処理
     *
     * @param string $email
     * @param string $password
     * @return bool 成功したら true、失敗したら false
     */
    public function login(string $email, string $password): bool {
        $stmt = $this->pdo->prepare(
            'SELECT id, username, email, password_hash, role FROM users WHERE email = :email LIMIT 1'
        );
        $stmt->execute(['email' => $email]);
        $user = $stmt->fetch();

        if (!$user) {
            $_SESSION['login_error'] = 'メールアドレスまたはパスワードが正しくありません。';
            return false;
        }

        if (!verify_password($password, $user['password_hash'])) {
            $_SESSION['login_error'] = 'メールアドレスまたはパスワードが正しくありません。';
            return false;
        }

        // パスワード OK → セッション開始
        $_SESSION['user_id']      = $user['id'];
        $_SESSION['username']     = $user['username'];
        $_SESSION['email']        = $user['email'];
        $_SESSION['role']         = $user['role']; // admin / member
        $_SESSION['csrf']         = bin2hex(random_bytes(32));

        // ログアウト時に削除されるようタイムスタンプ更新
        $_SESSION['last_activity'] = time();

        return true;
    }

    /**
     * 会員登録処理
     *
     * @param string $username
     * @param string $email
     * @param string $password
     * @return bool 成功したら true、失敗したら false
     */
    public function register(string $username, string $email, string $password): bool {
        // 重複チェック
        $stmt = $this->pdo->prepare(
            'SELECT id FROM users WHERE email = :email OR username = :username'
        );
        $stmt->execute(['email' => $email, 'username' => $username]);
        if ($stmt->fetch()) {
            $_SESSION['register_error'] = 'このメールアドレスまたはユーザー名はすでに使用されています。';
            return false;
        }

        // パスワードハッシュ化
        $hash = hash_password($password);

        // INSERT
        $stmt = $this->pdo->prepare(
            'INSERT INTO users (username, email, password_hash, role) VALUES (:u, :e, :p, :r)'
        );
        $stmt->execute([
            'u' => $username,
            'e' => $email,
            'p' => $hash,
            'r' => 'member',
        ]);

        // 任意：歓迎メール送信
        $subject = '【'.APP_NAME.'】会員登録完了のお知らせ';
        $body = "こんにちは {$username} さん、\n\n".
                "このたびは『".APP_NAME."』にご登録いただきありがとうございます。\n".
                "何かご不明点がございましたら、{$ADMIN_EMAIL} までご連絡ください。\n\n".
                "―――\n".
                "{$ADMIN_EMAIL}";
        send_mail($email, $subject, $body);

        // ログイン状態へ遷移させる
        $stmt = $this->pdo->prepare(
            'SELECT id, username, email, password_hash, role FROM users WHERE email = :email LIMIT 1'
        );
        $stmt->execute(['email' => $email]);
        $user = $stmt->fetch();
        $_SESSION['user_id']      = $user['id'];
        $_SESSION['username']     = $user['username'];
        $_SESSION['email']        = $user['email'];
        $_SESSION['role']         = $user['role'];
        $_SESSION['csrf']         = bin2hex(random_bytes(32));
        $_SESSION['last_activity']= time();

        return true;
    }

    /**
     * ログアウト処理
     */
    public function logout(): void {
        $_SESSION = [];
        // Cookie の削除
        if (ini_get("session.use_cookies")) {
            $params = session_get_cookie_params();
            setcookie(session_name(), '', time() - 42000,
                $params["path"], $params["domain"],
                $params["secure"], $params["httponly"]
            );
        }
        session_destroy();
    }

    /**
     * 現在のユーザーがログイン中か判定
     *
     * @return bool
     */
    public function isLoggedIn(): bool {
        return !empty($_SESSION['user_id']);
    }

    /**
     * 現在のユーザー名取得（未ログイン時は null）
     *
     * @return string|null
     */
    public function getUsername(): ?string {
        return $_SESSION['username'] ?? null;
    }

    /**
     * ロール取得（admin / member / guest）
     *
     * @return string
     */
    public function getRole(): string {
        return $_SESSION['role'] ?? 'guest';
    }
}

/* ---------- インスタンス生成 ----------
   以降のページでは $auth という変数で利用できます。
*/
$auth = new AuthService($GLOBALS['pdo']);
```

```php
<?php
declare(strict_types=1);
require_once __DIR__.'/config.php';
require_once __DIR__.'/functions.php';
require_once __DIR__.'/auth.php';

// CSRF チェック
if (empty($_SESSION['csrf']) || !check_csrf($_POST['csrf'] ?? '')) {
    $_SESSION['register_error'] = '不正なリクエストです。';
    header('Location: '.BASE_URL.'register.php');
    exit;
}

// POST 受信 → 登録処理
if ($_SERVER['REQUEST_METHOD'] === 'POST') {
    $username = trim($_POST['username'] ?? '');
    $email    = trim($_POST['email'] ?? '');
    $password = $_POST['password'] ?? '';

    if (strlen($username) < 3 || strlen($username) > 20) {
        $_SESSION['register_error'] = 'ユーザー名は3〜20文字で入力してください。';
    } elseif (!filter_var($email, FILTER_VALIDATE_EMAIL)) {
        $_SESSION['register_error'] = '有効なメールアドレスを入力してください。';
    } elseif (strlen($password) < 6) {
        $_SESSION['register_error'] = 'パスワードは6文字以上で入力してください。';
    } else {
        // 認証サービスが登録を実施
        if ($auth->register($username, $email, $password)) {
            header('Location: '.BASE_URL.'home.php');
            exit;
        }
    }
}

$title = '会員登録';
$view  = 'views/register.php';
require __DIR__.'/template.php';
```

## 9️⃣ `public/models/`（例：`BookModel.php`）

```php
<?php
declare(strict_types=1);

/**
 * 本（books）テーブル操作用クラス
 */
class BookModel {
    private PDO $pdo;

    public function __construct(PDO $pdo) {
        $this->pdo = $pdo;
    }

    /** 全件取得（ページング対応） */
    public function getAll(int $offset = 0, int $limit = 20): array {
        $stmt = $this->pdo->prepare(
            'SELECT id, title, author, published_year, isbn, status
             FROM books ORDER BY published_year DESC, title ASC LIMIT :limit OFFSET :offset'
        );
        $stmt->execute(['limit' => $limit, 'offset' => $offset]);
        return $stmt->fetchAll();
    }

    /** キーワード検索 */
    public function search(string $keyword, int $offset = 0, int $limit = 20): array {
        $kw = "%{$keyword}%";
        $stmt = $this->pdo->prepare(
            'SELECT id, title, author, published_year, isbn, status
             FROM books
             WHERE title LIKE :kw OR author LIKE :kw
             ORDER BY title ASC
             LIMIT :limit OFFSET :offset'
        );
        $stmt->execute(['kw' => $kw, 'limit' => $limit, 'offset' => $offset]);
        return $stmt->fetchAll();
    }

    /** ISBN で在庫取得（貸し出し・返却時に使用） */
    public function getById(int $bookId): ?array {
        $stmt = $this->pdo->prepare(
            'SELECT id, title, author, published_year, isbn, status
             FROM books WHERE id = :id LIMIT 1'
        );
        $stmt->execute(['id' => $bookId]);
        return $stmt->fetch();
    }

    /** 貸し出しステータス更新 */
    public function setOnLoan(int $bookId, int $userId): bool {
        $stmt = $this->pdo->prepare(
            'UPDATE books SET status = :status, loaned_to = :uid, loaned_at = NOW()
             WHERE id = :id AND status = "available"'
        );
        return $stmt->execute([
            'status' => 'loaned',
            'uid'    => $userId,
            'id'     => $bookId,
        ]);
    }

    /** 返却ステータス更新 */
    public function setReturned(int $bookId): bool {
        $stmt = $this->pdo->prepare(
            'UPDATE books SET status = "available", loaned_to = NULL, loaned_at = NULL
             WHERE id = :id AND status = "loaned"'
        );
        return $stmt->execute(['id' => $bookId]);
    }
}
```

```php
    $bookId = intval($_POST['book_id'] ?? 0);
    $userId = intval($_SESSION['user_id'] ?? 0);
    if ($bookId && $userId) {
        $model = new BookModel($GLOBALS['pdo']);
        if ($model->setOnLoan($bookId, $userId)) {
            $_SESSION['loan_success'] = '貸し出しが完了しました。';
        } else {
            $_SESSION['loan_error'] = 'その書籍は現在貸し出されています。';
        }
    } else {

}

exit;
```

# 🎯 次のステップ（実装ヒント）

| 項目 | 推奨実装方法 |
|------|--------------|
| **ページング** | `search.php` に `page` パラメータを導入し、`LIMIT offset, limit` で分割表示。 |
| **CSRF トークン自動埋め込み** | `template.php` の `<form>` 部分で `<input type="hidden" name="csrf" value="<?= $_SESSION['csrf'] ?>">` を自動出力。 |
| **バリデーション強化** | `filter_var`, 正規表現, `password_needs_rehash()` でパスワードポリシーをチェック。 |
| **メール送信** | `send_mail()` を `register.php` と `login.php` の成功後に呼び出す（認証リンク方式にすれば更に安全）。 |
| **管理者機能** | `role = admin` のユーザだけがアクセスできる `admin/*` ディレクトリを作り、書籍の追加・削除・在庫調整 API を実装。 |
| **テスト** | PHPUnit で `BookModel` や `AuthService` の単体テストを書き、CI（GitHub Actions）で自動走査。 |
| **デプロイ** | Dockerfile を作成し、`docker-compose.yml` で MySQL と PHP-FPM を同梱すれば、本番環境でも同一コードで動作。 |
